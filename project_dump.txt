"""
Синтетический Blackbox-пробер для сервиса sre-mentorflow.
Опрашивает UI (/login) и REST API (/api/v1/users), проверяя HTTP-статус и время ответа <= 250 мс.
"""

import time
import threading
import urllib.request
import urllib.error
from http.server import BaseHTTPRequestHandler, HTTPServer

TARGET_BASE = "http://mentorflow-service.mentorflow.svc.cluster.local:8080"
TIMEOUT_SEC = 0.5
LATENCY_SLO_SEC = 0.25

metrics_lock = threading.Lock()
state = {
    "probes_total": {"ui": 0, "api": 0},
    "probes_success_total": {"ui": 0, "api": 0},
    "last_status": {"ui": 1, "api": 1},
    "last_duration_seconds": {"ui": 0.01, "api": 0.01},
}


def check_endpoint(name: str, path: str) -> None:
    url = f"{TARGET_BASE}{path}"
    start = time.time()
    success = 0
    try:
        req = urllib.request.Request(url, method="GET")
        with urllib.request.urlopen(req, timeout=TIMEOUT_SEC) as resp:
            duration = time.time() - start
            if 200 <= resp.status < 400 and duration <= LATENCY_SLO_SEC:
                success = 1
    except Exception:
        duration = time.time() - start
        success = 0

    with metrics_lock:
        state["probes_total"][name] += 1
        if success == 1:
            state["probes_success_total"][name] += 1
        state["last_status"][name] = success
        state["last_duration_seconds"][name] = round(duration, 4)


def prober_loop() -> None:
    while True:
        check_endpoint("ui", "/login")
        check_endpoint("api", "/api/v1/users")
        time.sleep(2)


class MetricsHandler(BaseHTTPRequestHandler):
    def do_GET(self) -> None:
        if self.path != "/metrics":
            self.send_response(404)
            self.end_headers()
            return

        lines = [
            "# HELP mentorflow_prober_requests_total Total synthetic probes executed",
            "# TYPE mentorflow_prober_requests_total counter",
            "# HELP mentorflow_prober_success_total Total successful synthetic probes",
            "# TYPE mentorflow_prober_success_total counter",
            "# HELP mentorflow_prober_last_status Last probe status (1=up, 0=down)",
            "# TYPE mentorflow_prober_last_status gauge",
            "# HELP mentorflow_prober_duration_seconds Last probe duration in seconds",
            "# TYPE mentorflow_prober_duration_seconds gauge",
        ]

        with metrics_lock:
            for check in ("ui", "api"):
                lines.append(
                    f'mentorflow_prober_requests_total{{check="{check}"}} {state["probes_total"][check]}'
                )
                lines.append(
                    f'mentorflow_prober_success_total{{check="{check}"}} {state["probes_success_total"][check]}'
                )
                lines.append(
                    f'mentorflow_prober_last_status{{check="{check}"}} {state["last_status"][check]}'
                )
                lines.append(
                    f'mentorflow_prober_duration_seconds{{check="{check}"}} {state["last_duration_seconds"][check]}'
                )

        body = ("\n".join(lines) + "\n").encode("utf-8")
        self.send_response(200)
        self.send_header("Content-Type", "text/plain; version=0.0.4; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def log_message(self, format: str, *args) -> None:
        pass


if __name__ == "__main__":
    worker = threading.Thread(target=prober_loop, daemon=True)
    worker.start()
    server = HTTPServer(("0.0.0.0", 8000), MetricsHandler)
    server.serve_forever()

package com.mukminov.chaos;

import com.mukminov.metrics.MentorFlowMetrics;
import jakarta.servlet.FilterChain;
import jakarta.servlet.ServletException;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import lombok.RequiredArgsConstructor;
import org.springframework.core.Ordered;
import org.springframework.core.annotation.Order;
import org.springframework.stereotype.Component;
import org.springframework.web.filter.OncePerRequestFilter;

import java.io.IOException;
import java.util.concurrent.ThreadLocalRandom;

/**
 * HTTP-фильтр симуляции задержек и ошибок HTTP 500 для демонстрации SLI/SLO/SLA.
 */
@Component
@Order(Ordered.HIGHEST_PRECEDENCE + 10)
@RequiredArgsConstructor
public class ChaosSimulationFilter extends OncePerRequestFilter {

    private final ChaosState chaosState;
    private final MentorFlowMetrics mentorFlowMetrics;

    @Override
    protected boolean shouldNotFilter(HttpServletRequest request) {
        String path = request.getRequestURI();
        return path.startsWith("/actuator") || path.startsWith("/api/v1/chaos");
    }

    @Override
    protected void doFilterInternal(HttpServletRequest request,
                                    HttpServletResponse response,
                                    FilterChain filterChain) throws ServletException, IOException {

        // Искусственная задержка (Latency SLI)
        int delay = chaosState.getLatencyMs();
        if (delay > 0) {
            try {
                Thread.sleep(delay);
            } catch (InterruptedException e) {
                Thread.currentThread().interrupt();
            }
        }

        // Демонстрационная фиксация таймера GitHub Sync
        if (chaosState.isDegradeBusinessMetrics()) {
            mentorFlowMetrics.recordGitHubSyncDuration(1500L);
        } else {
            mentorFlowMetrics.recordGitHubSyncDuration(80L);
        }

        // Инжекция HTTP 500 (Availability SLI)
        int errorRate = chaosState.getErrorRatePercent();
        if (errorRate > 0 && ThreadLocalRandom.current().nextInt(100) < errorRate) {
            response.setStatus(HttpServletResponse.SC_INTERNAL_SERVER_ERROR);
            response.setContentType("application/json");
            response.getWriter().write("{\"error\":true,\"message\":\"Chaos: Simulated 500 Error\"}");
            return;
        }

        filterChain.doFilter(request, response);
    }
}

package com.mukminov.client;

import com.fasterxml.jackson.databind.JsonNode;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.mukminov.properties.GitHubClientProperties;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import okhttp3.OkHttpClient;
import okhttp3.Request;
import okhttp3.Response;
import org.springframework.cache.annotation.Cacheable;
import org.springframework.stereotype.Component;

import java.io.IOException;

@Slf4j
@Component
@RequiredArgsConstructor
public class GitHubClient {

    private final OkHttpClient okHttpClient;
    private final ObjectMapper objectMapper;
    private final GitHubClientProperties properties;

    @Cacheable(value = "githubCommits", key = "#owner + ':' + #repo + ':' + #githubUsername")
    public int getCommitsCount(String owner, String repo, String githubUsername, String sinceIso) {
        String url = String.format("https://api.github.com/repos/%s/%s/commits?since=%s&author=%s",
                owner, repo, sinceIso, githubUsername);

        log.debug("Sending request to GitHub API: {}", url);

        Request request = new Request.Builder()
                .url(url)
                .header("Accept", "application/vnd.github+json")
                .header("User-Agent", "MentorFlow-App")
                .header("Authorization", "Bearer " + properties.getToken())
                .build();

        try (Response response = okHttpClient.newCall(request).execute()) {
            if (!response.isSuccessful()) {
                log.warn("GitHub API returned status {} for repository {}/{}", response.code(), owner, repo);
                return 0;
            }

            String responseBody = response.body() != null ? response.body().string() : "[]";
            JsonNode rootArray = objectMapper.readTree(responseBody);

            if (rootArray.isArray()) {
                int count = rootArray.size();
                log.debug("Found {} commits for user {} since {}", count, githubUsername, sinceIso);
                return count;
            }
        } catch (IOException e) {
            log.error("Network error while calling GitHub API for {}/{}", owner, repo, e);
        }

        return 0;
    }
}

package com.mukminov.exception;

import io.swagger.v3.oas.annotations.Hidden;
import jakarta.servlet.http.HttpServletRequest;
import lombok.extern.slf4j.Slf4j;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.ControllerAdvice;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.servlet.ModelAndView;
import org.springframework.web.servlet.resource.NoResourceFoundException;

import java.util.Map;

@Slf4j
@Hidden
@ControllerAdvice
public class GlobalExceptionHandler {

    @ExceptionHandler(NoResourceFoundException.class)
    public Object handleResourceNotFound(NoResourceFoundException ex, HttpServletRequest request) {
        if (isAjax(request)) {
            return ResponseEntity.status(HttpStatus.NOT_FOUND)
                    .body(Map.of(
                            "error", true,
                            "message", "Ресурс не найден"
                    ));
        }

        ModelAndView mav = new ModelAndView("error/404");
        mav.setStatus(HttpStatus.NOT_FOUND);
        return mav;
    }

    @ExceptionHandler(Exception.class)
    public Object handleAllExceptions(Exception ex, HttpServletRequest request) {
        log.error("Unhandled exception caught: ", ex);

        if (isAjax(request)) {
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR)
                    .body(Map.of(
                            "error", true,
                            "message", "Произошла внутренняя ошибка сервера. Попробуйте повторить позже"
                    ));
        }

        ModelAndView modelAndView = new ModelAndView("error/500");
        modelAndView.addObject("errorMessage", ex.getMessage());
        modelAndView.setStatus(HttpStatus.INTERNAL_SERVER_ERROR);
        return modelAndView;
    }

    private boolean isAjax(HttpServletRequest request) {
        String requestedWith = request.getHeader("X-Requested-With");
        String accept = request.getHeader("Accept");

        return "XMLHttpRequest".equals(requestedWith) ||
                (accept != null && accept.contains("application/json"));
    }
}

import com.mukminov.repository.RoadmapRepository;
import io.micrometer.core.instrument.Counter;

import io.micrometer.core.instrument.Timer;

import org.springframework.security.authentication.event.AbstractAuthenticationFailureEvent;
import org.springframework.security.authentication.event.AuthenticationSuccessEvent;
import org.springframework.stereotype.Component;

import java.util.concurrent.TimeUnit;

@Component
public class MentorFlowMetrics {

    private final Counter usersRegisteredCounter;
    private final Counter authSuccessCounter;
    private final Counter authFailureCounter;
    private final Counter roadmapsCreatedCounter;
    private final Counter stepsCompletedCounter;
    private final Counter matchesCreatedCounter;
    private final Timer githubSyncTimer;

    public MentorFlowMetrics(MeterRegistry registry, RoadmapRepository roadmapRepository) {
        // 1. Количество зарегистрированных пользователей
        this.usersRegisteredCounter = Counter.builder("mentorflow_users_registered_total")
                .description("Total number of registered users in MentorFlow")
                .register(registry);

        // 2. Попытки авторизации (с разделением на успешные и неуспешные)
        this.authSuccessCounter = Counter.builder("mentorflow_auth_attempts_total")
                .tag("status", "success")
                .description("Total number of authentication attempts")
                .register(registry);

        this.authFailureCounter = Counter.builder("mentorflow_auth_attempts_total")
                .tag("status", "failure")
                .description("Total number of authentication attempts")
                .register(registry);

        // 3. Количество созданных роадмапов обучения
        this.roadmapsCreatedCounter = Counter.builder("mentorflow_roadmaps_created_total")
                .description("Total number of created mentorship roadmaps")
                .register(registry);

        // 4. Количество успешно завершённых шагов роадмапа (статус DONE)
        this.stepsCompletedCounter = Counter.builder("mentorflow_roadmap_steps_completed_total")
                .description("Total number of completed roadmap steps")
                .register(registry);

        // 5. Количество успешных мэтчей ментор-менти на доске объявлений
        this.matchesCreatedCounter = Counter.builder("mentorflow_matches_created_total")
                .description("Total number of accepted mentor-mentee connection requests")
                .register(registry);

        // 6. Текущее количество активных роадмапов в базе данных
        Gauge.builder("mentorflow_active_roadmaps_count", roadmapRepository, RoadmapRepository::count)
                .description("Current number of active roadmaps in database")
                .register(registry);

        // 7. Таймер длительности синхронизации коммитов с GitHub API
        this.githubSyncTimer = Timer.builder("mentorflow_github_sync_duration_seconds")
                .description("Duration of GitHub commits synchronization")
                .register(registry);
    }

    // Автоматический перехват успешных логинов через Spring Security
    @EventListener
    public void onAuthSuccess(AuthenticationSuccessEvent event) {
        authSuccessCounter.increment();
    }

    // Автоматический перехват неудачных логинов через Spring Security
    @EventListener
    public void onAuthFailure(AbstractAuthenticationFailureEvent event) {
        authFailureCounter.increment();
    }

    public void incrementUsersRegistered() {
        usersRegisteredCounter.increment();
    }

    public void incrementRoadmapsCreated() {
        roadmapsCreatedCounter.increment();
    }

    public void incrementStepsCompleted() {
        stepsCompletedCounter.increment();
    }

    public void incrementMatchesCreated() {
        matchesCreatedCounter.increment();
    }

    public void recordGitHubSyncDuration(long durationMs) {
        githubSyncTimer.record(durationMs, TimeUnit.MILLISECONDS);
    }
}

package com.mukminov.repository.impl;

import com.mukminov.entity.Roadmap;
import com.mukminov.repository.RoadmapRepositoryCustom;
import jakarta.persistence.EntityManager;
import jakarta.persistence.PersistenceContext;
import jakarta.persistence.criteria.*;
import org.springframework.stereotype.Repository;

import java.util.ArrayList;
import java.util.List;

@Repository
public class RoadmapRepositoryCustomImpl implements RoadmapRepositoryCustom {

    @PersistenceContext
    private EntityManager entityManager;

    @Override
    public List<Roadmap> searchRoadmaps(String title, Long mentorId, Long menteeId) {
        CriteriaBuilder criteriaBuilder = entityManager.getCriteriaBuilder();
        CriteriaQuery<Roadmap> query = criteriaBuilder.createQuery(Roadmap.class);
        Root<Roadmap> root = query.from(Roadmap.class);

        root.fetch("mentor", JoinType.INNER);
        root.fetch("mentee", JoinType.INNER);

        List<Predicate> predicates = new ArrayList<>();

        if (title != null && !title.isBlank()) {
            predicates.add(criteriaBuilder.like(criteriaBuilder.lower(root.get("title")), "%" + title.toLowerCase() + "%"));
        }
        
        if (mentorId != null) {
            predicates.add(criteriaBuilder.equal(root.get("mentor").get("id"), mentorId));
        }
        
        if (menteeId != null) {
            predicates.add(criteriaBuilder.equal(root.get("mentee").get("id"), menteeId));
        }

        query.where(criteriaBuilder.and(predicates.toArray(new Predicate[0])));

        query.orderBy(criteriaBuilder.desc(root.get("createdAt")));

        return entityManager.createQuery(query).getResultList();
    }
}

                .orElseThrow(() -> new RuntimeException("Advertisement not found"));
        User sender = userRepository.findById(dto.getSenderId())
                .orElseThrow(() -> new RuntimeException("Sender not found"));

        boolean alreadySent = requestRepository.existsBySenderIdAndAdvertisementId(dto.getSenderId(), adId);
        if (alreadySent) {
            throw new RuntimeException("Вы уже отправляли отклик на эту заявку!");
        }

        ConnectionRequest req = ConnectionRequest.builder()
                .advertisement(ad)
                .sender(sender)
                .coverLetter(dto.getCoverLetter())
                .status(ConnectionRequest.RequestStatus.PENDING)
                .build();

        return reqMapper.toDto(requestRepository.save(req));
    }

    @Override
    @Transactional(readOnly = true)
    public List<ConnectionRequestDto> getIncomingRequests(Long userId) {

                .stream()
                .map(reqMapper::toDto)
                .toList();
    }

    @Override
    @Transactional
    public ConnectionRequestDto updateRequestStatus(Long requestId, ConnectionRequestStatusUpdateDto dto) {
        ConnectionRequest req = requestRepository.findById(requestId)
                .orElseThrow(() -> new RuntimeException("Request not found"));
        
        ConnectionRequest.RequestStatus newStatus = ConnectionRequest.RequestStatus.valueOf(dto.getStatus());
        req.setStatus(newStatus);

        if (newStatus == ConnectionRequest.RequestStatus.ACCEPTED) {
            Advertisement ad = req.getAdvertisement();
            Long mentorId;
            Long menteeId;

            if (ad.getType() == Advertisement.AdType.SEEKING_MENTOR) {
                menteeId = ad.getAuthor().getId();
                mentorId = req.getSender().getId();
            } else {
                mentorId = ad.getAuthor().getId();
                menteeId = req.getSender().getId();
            }

            userRepository.addConnection(mentorId, menteeId);
            log.info("Match created! Mentor ID: {}, Mentee ID: {}", mentorId, menteeId);
            metrics.incrementMatchesCreated();

            ad.setStatus(Advertisement.AdStatus.CLOSED);
            adRepository.save(ad);
        }

        return reqMapper.toDto(requestRepository.save(req));
    }

        RoadmapStep step = stepRepository.findById(stepId)
                .orElseThrow(() -> new RuntimeException("Шаг не найден"));

        if (step.getStatus() == RoadmapStep.StepStatus.IN_PROGRESS) {
            processStep(step);
        }
        return stepMapper.toDto(step);
    }

    private void processStep(RoadmapStep step) {
        String repoUrl = step.getContentLink();
        String githubUsername = step.getRoadmap().getMentee().getGithubUsername();

        if (githubUsername == null || githubUsername.isEmpty() || step.getStartedAt() == null) {
            return;
        }

        String[] ownerAndRepo = GitHubUrlParser.extractOwnerAndRepo(repoUrl);
        if (ownerAndRepo == null) return;

        String owner = ownerAndRepo[0];
        String repo = ownerAndRepo[1];
        String sinceIso = step.getStartedAt().atOffset(ZoneOffset.UTC)
                .format(DateTimeFormatter.ISO_INSTANT);

        int actualCommits = gitHubClient.getCommitsCount(owner, repo, githubUsername, sinceIso);

        log.info("Step ID: {} | Required: {} | Found: {}", step.getId(), step.getRequiredCommits(), actualCommits);

        step.setActualCommits(actualCommits);

        if (actualCommits >= step.getRequiredCommits()) {
            log.info("Step ID: {} completed required commits. Moving to REVIEW.", step.getId());
            step.setStatus(RoadmapStep.StepStatus.REVIEW);
        } else {
            int maxAfkDays = step.getRoadmap().getMaxAfkDays() != null ? step.getRoadmap().getMaxAfkDays() : 5;
            String meetLink = step.getRoadmap().getMeetLink() != null ? step.getRoadmap().getMeetLink() : "https://meet.google.com/new";

            if (step.getStartedAt().plusDays(maxAfkDays).isBefore(LocalDateTime.now())) {
                Long menteeId = step.getRoadmap().getMentee().getId();

                boolean hasActiveConsultation = consultationRepository.existsByMenteeIdAndStatus(
                        menteeId, Consultation.ConsultationStatus.SCHEDULED);

                if (!hasActiveConsultation) {
                    log.warn("Mentee {} is slacking ({} days). Auto-scheduling intervention!", githubUsername, maxAfkDays);

                    Consultation intervention = Consultation.builder()
                            .mentor(step.getRoadmap().getMentor())
                            .mentee(step.getRoadmap().getMentee())
                            .scheduledTime(LocalDateTime.now().plusDays(1))
                            .meetLink(meetLink)
                            .status(Consultation.ConsultationStatus.SCHEDULED)
                            .build();

                    consultationRepository.save(intervention);
                }
            }
        }

        stepRepository.save(step);
    }
}

package com.mukminov.util;

import lombok.experimental.UtilityClass;
import lombok.extern.slf4j.Slf4j;

import java.net.URI;

@Slf4j
@UtilityClass
public class GitHubUrlParser {

    public static String[] extractOwnerAndRepo(String url) {
        try {
            String path = URI.create(url).getPath();

            String[] parts = path.split("/");
            if (parts.length >= 3) {
                return new String[]{parts[1], parts[2].replace(".git", "")};
            }
        } catch (Exception e) {
            log.error("Error parsing GitHub URL: {}", url, e);
        }
        return null;
    }
}

import time
import threading
import urllib.request

TARGET_URLS = [
    "http://localhost:8080/login",
    "http://localhost:8080/api/v1/users",
]
TARGET_RPS = 60
RUNNING = True


def worker():
    while RUNNING:
        start = time.time()
        for url in TARGET_URLS:
            try:
                with urllib.request.urlopen(url, timeout=1.5) as resp:
                    resp.read()
            except Exception:
                pass
        elapsed = time.time() - start
        sleep_sec = max(0.0, (len(TARGET_URLS) * 6 / TARGET_RPS) - elapsed)
        if sleep_sec > 0:
            time.sleep(sleep_sec)


if __name__ == "__main__":
    print(f"Генератор нагрузки запущен (~{TARGET_RPS} RPS). Нажмите Ctrl+C для выхода.")
    for _ in range(6):
        threading.Thread(target=worker, daemon=True).start()
    try:
        while True:
            time.sleep(1)
    except KeyboardInterrupt:
        RUNNING = False

# MentorFlow

Платформа для организации IT-менторства. Помогает находить менторов и менти через доску объявлений, планировать консультации и выстраивать процесс обучения с помощью дорожных карт с интеграцией коммитов из GitHub.

Проект переработан в соответствии с **методологией 12 факторов** (Cloud Native) для курса SRE:
* Приложение работает как полностью stateless-процесс.
* Конфигурация вынесена в переменные окружения.
* Настроены healthchecks для всех компонентов инфраструктуры.
* Реализован graceful shutdown.

## 🛠 Стек технологий

* **Backend:** Java 21, Spring Boot 3.5.14
* **База данных:** PostgreSQL 16 (миграции через Flyway)
* **Кэш:** Redis 7
* **Frontend:** Freemarker (SSR)
* **Инфраструктура:** Docker, Docker Compose, Kubernetes (Minikube)

## 🚀 Как запустить локально (через Docker Compose)

Для запуска приложения на вашей машине потребуются установленные **Docker** и **Docker Compose**.

### 1. Подготовка конфигурации

В корне проекта создайте файл `.env` на основе предоставленного шаблона `.env_dist`.

В Linux/macOS это можно сделать командой:
```bash
cp .env_dist .env
```

Обязательно откройте файл `.env` и подставьте ваш действующий токен GitHub в переменную `GITHUB_TOKEN`, иначе функционал синхронизации шагов дорожной карты не будет работать:

```env
# Настройки PostgreSQL
POSTGRES_USER=postgres
POSTGRES_PASSWORD=postgres
POSTGRES_DB=mentorflow_db
POSTGRES_URL=jdbc:postgresql://db:5432/mentorflow_db

# Токен для интеграции с GitHub API
GITHUB_TOKEN=ваш_реальный_токен_здесь

# Настройки Redis
REDIS_HOST=redis
REDIS_PORT=6379
```

### 2. Запуск инфраструктуры

Выполните сборку и запуск всех контейнеров в фоновом режиме:

```bash
docker-compose up -d --build
```

Дождитесь, пока все контейнеры перейдут в статус `healthy` (база данных применит миграции Flyway, а приложение успешно стартует)

### 3. Доступ к сервисам

После успешного запуска сервисы будут доступны по следующим адресам:

* **Веб-интерфейс приложения:** [http://localhost:8080](http://localhost:8080)
* **Проверка состояния (Healthcheck):** [http://localhost:8080/actuator/health](http://localhost:8080/actuator/health)
* **База данных PostgreSQL:** `localhost:5433`
* **Кэш и сессии Redis:** `localhost:6379`

### Остановка приложения в Docker

Чтобы корректно остановить приложение и удалить контейнеры (данные в БД сохранятся благодаря volume), выполните:

```bash
docker-compose down
```

## ☸️ Как запустить в Kubernetes (Minikube)

Проект полностью готов к развертыванию в кластере Kubernetes. Манифесты настроены с учетом сохранения состояния базы данных (StatefulSet + PersistentVolumeClaim)

1. Укажите ваш токен GitHub в файле `k8s/config.yml`.
2. Примените все манифесты одной командой:

```bash
kubectl apply -k k8s/
```

3. Дождитесь перехода всех подов в статус `Running` (проверить можно командой `kubectl get pods -w`).
4. Для доступа к веб-интерфейсу пробросьте порт наружу с помощью встроенной утилиты Minikube:

```bash
minikube service mentorflow-service -n mentorflow
```

Для удаления всех ресурсов кластера выполните: `kubectl delete -k k8s/`

## 🧩 Добавление нового функционала

Проект построен по классическому слоистому паттерну Spring Boot. Чтобы добавить новую логику:

1. **API**: Опишите эндпоинты в `openapi.yaml` или создайте контроллер в пакете `controller`.
2. **Бизнес-логика**: Реализуйте интерфейс и класс сервиса в пакете `service`.
3. **БД**: Добавьте сущность в `entity` и интерфейс в `repository`, а также напишите новую миграцию в `db/migration`.

### Локальный профиль (In-memory сессии)

При необходимости хранение HTTP-сессий можно переключить из Redis в локальную память (in-memory). Для этого достаточно активировать профиль `local`, добавив настройку в файл `.env`:

```env
SPRING_PROFILES_ACTIVE=local
```

# Отчёт по домашнему заданию SRE: Приложение 12 факторов

## Доменная область приложения
**MentorFlow** — это платформа для организации IT-менторства. Приложение помогает пользователям (начинающим и опытным специалистам) находить друг друга через доску объявлений, договариваться о консультациях и выстраивать процесс обучения через дорожные карты (Roadmaps) с отслеживанием прогресса по шагам и ревью кода.

## Стек реализации
* **Backend:** Java 21, Spring Boot 3.5.14 (Web, Data JPA, Security, Cache)
* **База данных:** PostgreSQL 16 (миграции через Flyway)
* **Кэширование:** Redis 7
* **Frontend:** Шаблонизатор Freemarker (Server-Side Rendering)
* **Сборка и развертывание:** Gradle (Kotlin DSL), Docker, Docker Compose
* **Интеграции:** GitHub API (через OkHttp3)

## Основные сущности доменной области
1. **User** — пользователь платформы с ролями `ROLE_MENTOR` и/или `ROLE_MENTEE`.
2. **Advertisement** — объявление на доске о поиске ментора или менти.
3. **ConnectionRequest** — запрос на менторство (отклик на объявление с сопроводительным письмом).
4. **Consultation** — запланированная встреча (консультация) между ментором и менти.
5. **Roadmap** — дорожная карта развития, назначенная ментором для менти.
6. **RoadmapStep** — конкретный шаг/задача внутри дорожной карты.
7. **ReviewFeedback** — обратная связь от ментора по выполненному шагу.

## Аргументация соответствия методологии 12 факторов

1. **Кодовая база (Codebase):**
   Приложение представляет собой единую кодовую базу (отслеживается в Git, настроены `.gitignore` и `.gitattributes`), из которой собираются артефакты для локального запуска и продакшена.
2. **Зависимости (Dependencies):**
   Все зависимости явно и строго изолированно описаны в файле `build.gradle.kts` с фиксацией версий. Глобальные системные зависимости не используются, среда полностью изолирована внутри Docker-контейнера.
3. **Конфигурация (Config):**
   Конфигурация вынесена в переменные окружения. В `application.yaml` нет хардкода секретов или адресов (используются конструкции `${POSTGRES_URL}`, `${REDIS_HOST}`, `${GITHUB_TOKEN}`). Значения пробрасываются в контейнер через `.env` файл директивой `env_file` в `docker-compose.yml`.
4. **Сторонние службы (Backing services):**
   PostgreSQL и Redis рассматриваются как подключаемые ресурсы. Они поднимаются в отдельных контейнерах и подключаются к приложению по сети. Замена их на облачные Managed-решения потребует только изменения URL в переменных окружения.
5. **Сборка, релиз, выполнение (Build, release, run):**
   Процесс строго разделен благодаря multi-stage `Dockerfile`. На стадии `builder` исходный код компилируется в `.jar` артефакт. На стадии `runner` берется только JRE и скомпилированный артефакт для выполнения, что исключает возможность изменения кода на лету.
6. **Процессы (Processes):**
   Приложение выполняется как stateless-процесс. Сохранение состояния (данные пользователей, роадмапы) делегировано в PostgreSQL, а кэширование (данные из GitHub API) — во внешний Redis.
7. **Привязка портов (Port binding):**
   Приложение самодостаточно (self-contained) благодаря встроенному в Spring Boot серверу Tomcat. Сервис экспортируется наружу через привязку порта `8080` (указано в `EXPOSE` Докерфайла и `ports` в `docker-compose.yml`).
8. **Параллелизм (Concurrency):**
   Масштабирование происходит с помощью модели процессов. Благодаря stateless-архитектуре, можно легко запустить дополнительные экземпляры контейнера `mentorflow-app` для горизонтального масштабирования.
9. **Утилизируемость (Disposability):**
   Приложение готово к быстрому запуску и безопасному завершению. В `application.yaml` включена настройка `server.shutdown: graceful`, что позволяет корректно завершать обработку текущих HTTP-запросов при получении сигнала `SIGTERM` от Docker.
10. **Паритет окружений (Dev/prod parity):**
    Разработка ведется с использованием `docker-compose.yml`, что гарантирует абсолютную идентичность версий инфраструктуры (PostgreSQL 16-alpine, Redis 7-alpine) на локальной машине разработчика и на сервере.
11. **Журналирование (Logs):**
    Логи рассматриваются как непрерывный поток событий и пишутся в стандартный поток вывода (`stdout`). Приложение не управляет файлами `.log` на диске — сбор и ротацию логов осуществляет среда выполнения (Docker).
12. **Задачи администрирования (Admin processes):**
    Миграции базы данных автоматизированы с помощью инструмента Flyway (скрипты миграций лежат в `db/migration`). Они запускаются как неотъемлемая часть запуска приложения в том же окружении и с той же конфигурацией.

# Отчёт: Развёртывание MentorFlow в Kubernetes (Production-Ready)

## 1. Архитектура и объекты Kubernetes

В кластере Minikube развёрнуты следующие ресурсы (оркестрация через `k8s/kustomization.yaml`):

* **Namespace (`mentorflow`)**: все ресурсы приложения, базы данных и кэша изолированы от системных компонентов и `default`-неймспейса.
* **StatefulSet + PVC (`postgres-statefulset` и `redis-statefulset`)**: используются для обоих stateful-компонентов. PostgreSQL (`1Gi`) и Redis (`512Mi` с включённым AOF `--appendonly yes`) используют `volumeClaimTemplates`, что гарантирует сохранение состояния на постоянных томах (`PersistentVolumeClaim`) при перезапуске подов.
* **Job (`mentorflow-migration`)**: миграции схемы БД (Flyway) вынесены из старта веб-приложения в отдельную Kubernetes `Job` (`args: ["--migrate"]`) с `initContainer` ожидания готовности PostgreSQL. Само веб-приложение запускается с `SPRING_FLYWAY_ENABLED=false` только после успешного завершения миграций.
* **Deployment (`mentorflow-app`)**: управляет stateless-репликами бэкенда с поддержкой плавного завершения (`terminationGracePeriodSeconds: 30` + `preStop` хук) и автоскейлингом.
* **Probes (`startupProbe`, `readinessProbe`, `livenessProbe`)**: настроены для каждого компонента:
    * `postgres` — проверка готовности сокета через `pg_isready`;
    * `redis` — проверка ответа через `redis-cli ping`;
    * `mentorflow-app` — раздельные HTTP-пробы Spring Boot Actuator (`/actuator/health/liveness` и `/actuator/health/readiness`).
* **Resources (`requests` / `limits`)**: заданы для всех контейнеров.
* **Service (`ClusterIP` и `NodePort`)**: внутренние сервисы `postgres-service:5432` и `redis-service:6379` (`ClusterIP`), внешний доступ к UI приложения через `mentorflow-service` (`NodePort: 30080`).
* **ConfigMap и Secret**: `mentorflow-config` и `mentorflow-secret` хранят нечувствительные настройки и учётные данные.

## 2. Масштабирование и планирование 

* **HorizontalPodAutoscaler (`mentorflow-app-hpa`)**: настроен на автоматическое горизонтальное масштабирование `Deployment` от 1 до 3 реплик при превышении средней загрузки CPU порога в 70%

# Отчёт по домашнему заданию №2: Victoria Logs и централизованный сбор логов в Kubernetes

## 1. Архитектура решения и оркестрация через Kustomize

В кластере `minikube`, где работает приложение `sre-mentorflow` из ДЗ №1, развёрнут стек сбора, хранения и алертинга по логам:
1. **Изоляция в отдельном Namespace (`logging`)**: все компоненты наблюдаемости (`Victoria Logs`, `Vector`, `vmalert`) вынесены в собственное пространство имён `logging`.
2. **Декларативное управление конфигами без хардкода (`kustomization.yaml`)**: конфигурация агента `k8s/vector.toml` и правила алертинга `k8s/alerts.yml` вынесены в отдельные файлы и автоматически упаковываются в версионированные `ConfigMap` через `configMapGenerator` при выполнении команды `kubectl apply -k k8s/`
3. **Использование `StatefulSet` + `PVC`**: база данных `Victoria Logs` развёрнута как `StatefulSet` с `volumeClaimTemplates` (`1Gi`) для сохранности колоночных блоков (`parts`) при рестартах пода.
4. **Probes, Resources и Tolerations**: для всех новых компонентов настроены `requests`/`limits` по CPU и памяти, HTTP-пробы (`readinessProbe`, `livenessProbe` на `/health`), а для `DaemonSet` агента Vector добавлены `tolerations` для работы на узлах с `Taints`.

---

## 2. Настройка JSON-логирования в приложении (`sre-mentorflow`)

С помощью `logstash-logback-encoder` и конфигурации `src/main/resources/logback-spring.xml` логирование приложения переведено в структурированный формат JSON в `stdout`. В соответствии с чек-листом семинара и конвенциями OpenTelemetry каждая запись содержит поля:
* `timestamp` — время события в формате ISO 8601 (UTC)
* `level` — уровень важности (`INFO`, `WARN`, `ERROR`)
* `message` — текст сообщения
* `service.name` (`sre-mentorflow`), `service.version` (`1.0.0`), `deployment.environment.name` (`minikube`)
* `k8s.pod.name` — имя конкретного пода-реплики в Kubernetes
* `logger`, `thread` и `error.stack` (стектрейс исключения при ошибках)

---

## 3. Обоснование выбора способа доставки логов (Vector в режиме DaemonSet)

Для доставки логов из подов приложения в Victoria Logs выбран агент **Vector** (`timberio/vector:0.39.0-alpine`), развёрнутый как **`DaemonSet`** (`k8s/vector.yml` + `k8s/vector.toml`).

**Почему выбран именно этот способ:**
1. **Соблюдение 11-го фактора Twelve-Factor App:** приложение пишет поток событий в `stdout` и не зависит от доступности лог-хранилища.
2. **Высокая производительность и малый расход ресурсов:** Vector написан на Rust, не требует JVM и потребляет минимум CPU и RAM на ноде.
3. **Распаковка CRI/Docker-обёртки через VRL:** контейнерный рантайм в Kubernetes оборачивает строки `stdout` в технический формат CRI/Docker. С помощью VRL-трансформации (`remap`) Vector извлекает чистый JSON приложения перед отправкой.

**Нативный маппинг полей Victoria Logs:** отправка выполняется по протоколу Elasticsearch Bulk API (`/insert/elasticsearch/`) с параметрами `_msg_field = "message"`, `_time_field = "timestamp"` и `_stream_fields = "service.name,k8s.pod.name"`, `_level_field = "level"`, что обеспечивает группировку потоков и физическую локальность данных на диске

# Отчёт по домашнему заданию №4: Метрики

## 1. Описание стека приложения и системы мониторинга
* **Приложение:** `sre-mentorflow` — облачно-ориентированная веб-платформа для IT-менторства, управления индивидуальными планами обучения (Roadmaps), доски объявлений для поиска менторов/менти и автоматической синхронизации коммитов через GitHub API.
* **Backend-стек:** Java 21, Spring Boot 3.5.14 (Spring Web, Spring Data JPA, Spring Security, Spring Session Redis, Spring Boot Actuator), сборка на Gradle (Kotlin DSL).
* **Хранилища данных:** PostgreSQL 16 (миграции через Flyway в отдельном Kubernetes Job), Redis 7.
* **Оркестрация и инфраструктура:** Kubernetes (Minikube), управление манифестами через Kustomize (`k8s/kustomization.yaml`), приложение развёрнуто в выделенном namespace `mentorflow`.
* **Стек мониторинга (TSDB):** VictoriaMetrics Single (`vm-single-victoria-metrics-single-server`) и агент сбора метрик `vmagent` (`vm-single-victoria-metrics-agent`), развёрнутые через Helm в namespace `monitoring`.
* **Модель сбора метрик:** Pull-модель с автоматическим обнаружением таргетов (Kubernetes Service Discovery) при помощи pod-аннотаций `prometheus.io/scrape`, `prometheus.io/port` и `prometheus.io/path`.

## 2. Использованные экспортеры метрик
1. **Встроенный в приложение экспортер рантайма и фреймворка:**
    * Реализован с помощью библиотек `spring-boot-starter-actuator` и `io.micrometer:micrometer-registry-prometheus`.
    * Экспортирует метрики в формате OpenMetrics по HTTP-эндпоинту /actuator/prometheus на выделенном внутреннем порту 8081 (порт изолирован от внешнего NodePort 30080 и доступен без авторизации только внутри кластера)
    * Собираемые технические метрики: состояние памяти и сборщика мусора JVM (`jvm_memory_used_bytes`, `jvm_gc_*`, `jvm_threads_*`), загрузка CPU (`process_cpu_usage`, `system_cpu_usage`), метрики пула соединений БД (`hikaricp_connections_*`), метрики HTTP-запросов и гистограммы времени ответа (`http_server_requests_seconds_*`).
2. **Внешний экспортер СУБД PostgreSQL (Sidecar):**
    * Использован готовый экспортер `prometheuscommunity/postgres-exporter:v0.15.0`.
    * Развёрнут как sidecar-контейнер внутри пода `postgres-statefulset-0` (в общем сетевом пространстве пода), подключается к БД по `localhost:5432/mentorflow_db` и экспортирует метрики в формате OpenMetrics на порту `9187` по пути `/metrics` (`pg_up`, `pg_stat_database_*`, `pg_stat_activity_*`, `pg_locks_*`).

## 3. Реализованные бизнес-метрики приложения
В классе `com.mukminov.metrics.MentorFlowMetrics` с использованием `MeterRegistry` (Micrometer) реализованы следующие бизнес-метрики доменной области:

1. **`mentorflow_users_registered_total`** (`Counter`)
    * **Описание:** Общее количество зарегистрированных пользователей на платформе (инкрементируется в `UserServiceImpl.createUser`).
    * **Бизнес-смысл:** Мониторинг притока новых пользователей и контроль доступности процесса регистрации.
2. **`mentorflow_auth_attempts_total{status="success|failure"}`** (`Counter`)
    * **Описание:** Счётчик попыток входа в систему с разделением по тегу `status` (`success` при `AuthenticationSuccessEvent`, `failure` при `AbstractAuthenticationFailureEvent`).
    * **Бизнес-смысл:** Отслеживание пользовательской активности и обнаружение всплесков ошибок авторизации (сбои аутентификации или подбор паролей).
3. **`mentorflow_roadmaps_created_total`** (`Counter`)
    * **Описание:** Суммарное количество созданных менторами планов обучения (инкрементируется в `RoadmapServiceImpl.createRoadmap`).
    * **Бизнес-смысл:** Оценка продуктовой активности менторов по запуску новых программ обучения.
4. **`mentorflow_roadmap_steps_completed_total`** (`Counter`)
    * **Описание:** Количество успешно выполненных и принятых ментором шагов роадмапа со статусом `DONE` (инкрементируется в `ReviewFeedbackServiceImpl.addFeedbackToStep`).
    * **Бизнес-смысл:** Ключевая метрика образовательного прогресса менти на платформе.
5. **`mentorflow_matches_created_total`** (`Counter`)
    * **Описание:** Количество подтверждённых заявок (`ACCEPTED`) на доске объявлений, приведших к связке ментора и менти (`BoardServiceImpl.updateRequestStatus`).
    * **Бизнес-смысл:** Конверсия доски объявлений в реальные менторские пары.
6. **`mentorflow_active_roadmaps_count`** (`Gauge`)
    * **Описание:** Текущее количество активных роадмапов в базе данных (`RoadmapRepository::count`).
    * **Бизнес-смысл:** Отображение текущего объёма активных менторских сессий в реальном времени.
7. **`mentorflow_github_sync_duration_seconds`** (`Timer`)
    * **Описание:** Таймер времени выполнения фоновой и ручной синхронизации коммитов с GitHub API (`GitHubSyncServiceImpl`).
    * **Бизнес-смысл:** Контроль задержек (latency) и стабильности интеграции с внешним API GitHub.
